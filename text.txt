backend/
├── src/
│   ├── config/          # تنظیمات محیطی (Env vars)
│   ├── db/              # اتصال به دیتابیس و Redis
│   ├── routes/          # هندل کردن HTTP Routes (API Endpoints)
│   ├── handlers/        # توابع اصلی که درخواست را پردازش می‌کنند
│   ├── models/          # ساختار داده‌ها (DTOs) برای ورودی و خروجی JSON
│   ├── services/        # منطق اصلی بیزنس (Between Handler & DB)
│   ├── entities/        # مدل‌های دیتابیس (SeaORM Entities)
│   ├── middleware/      # میان‌افزارها (مثل JWT Auth)
│   ├── error/           # مدیریت خطاهای سفارشی
│   ├── utils/           # توابع کمکی
│   └── main.rs          # نقطه شروع برنامه
├── migration/           # فایل‌های مایگریشن دیتابیس
├── Cargo.toml           # وابستگی‌ها
└── Dockerfile
